cmake_minimum_required(VERSION 3.19)
project(trident LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT TRIDENT_CONFIG)
    message(FATAL_ERROR "TRIDENT_CONFIG is required")
endif()
if(NOT GGML_SRC)
    message(FATAL_ERROR "GGML_SRC is required")
endif()
if(NOT TRIDENT_UTF8_MANIFEST)
    message(FATAL_ERROR "TRIDENT_UTF8_MANIFEST is required")
endif()
if(NOT TRIDENT_MOUTH_COMPILE OR NOT TRIDENT_MOUTH_DEFINITIONS)
    message(FATAL_ERROR "TRIDENT_MOUTH_COMPILE and TRIDENT_MOUTH_DEFINITIONS are required")
endif()
if(NOT DEFINED MSVC_ARCH_FLAG)
    message(FATAL_ERROR "MSVC_ARCH_FLAG is required")
endif()
set(CMAKE_CONFIGURATION_TYPES "${TRIDENT_CONFIG}" CACHE STRING "" FORCE)
string(TOUPPER "${TRIDENT_CONFIG}" TRIDENT_CONFIG_UPPER)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${TRIDENT_CONFIG_UPPER} "${CMAKE_BINARY_DIR}/bin")
find_package(Vulkan REQUIRED)
add_subdirectory(${GGML_SRC} ${CMAKE_BINARY_DIR}/ggml)

add_library(trident_engine STATIC
    src/common/gguf_file.cpp src/common/vulkan_backend.cpp
    src/common/audio.cpp src/common/voice_encoder.cpp
    src/common/campplus.cpp src/common/s3_tokenizer.cpp src/common/s3_dsp.cpp
    src/common/s3.cpp src/common/chatterbox_runtime.cpp
    src/gpt2/t3.cpp src/gpt2/bpe.cpp src/gpt2/text_en.cpp src/gpt2/engine.cpp
    src/llama/t3.cpp src/llama/mtl_tokenizer.cpp src/llama/mtl_numbers.cpp src/llama/engine.cpp)
target_include_directories(trident_engine PUBLIC src ${GGML_SRC}/include)
target_link_libraries(trident_engine PUBLIC ggml Vulkan::Vulkan icu)

add_executable(chatterbox src/chatterbox.cpp)
add_executable(chatterbox-bake src/bake.cpp)
add_executable(ear src/ear.cpp)
if(NOT ONNXRUNTIME_DIR)
    message(FATAL_ERROR "ONNXRUNTIME_DIR is required")
endif()
add_executable(vad src/vad.cpp src/common/wasapi_capture.cpp src/common/silero_vad.cpp)
target_include_directories(ear PRIVATE src)
target_include_directories(vad PRIVATE src "${ONNXRUNTIME_DIR}/include")
target_link_libraries(chatterbox PRIVATE trident_engine)
target_link_libraries(chatterbox-bake PRIVATE trident_engine)
target_link_libraries(vad PRIVATE trident_engine "${ONNXRUNTIME_DIR}/lib/onnxruntime.lib" ole32 uuid)
separate_arguments(mouth_compile NATIVE_COMMAND "${TRIDENT_MOUTH_COMPILE}")
separate_arguments(mouth_definitions NATIVE_COMMAND "${TRIDENT_MOUTH_DEFINITIONS}")
foreach(target chatterbox chatterbox-bake ear vad)
    target_link_options(${target} PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${TRIDENT_UTF8_MANIFEST}")
endforeach()

foreach(target trident_engine chatterbox chatterbox-bake ear vad)
    target_compile_definitions(${target} PRIVATE ${mouth_definitions})
    target_compile_options(${target} PRIVATE ${mouth_compile})
    if(MSVC_ARCH_FLAG)
        target_compile_options(${target} PRIVATE ${MSVC_ARCH_FLAG})
    endif()
endforeach()
