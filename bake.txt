# chatterbox-bake.exe reads bake.t3, bake.s3, and bake.reference.
# bake.cond-seconds is the reference length. Nano uses 15. v3 uses 6.
bake.t3 nano-t3.gguf
bake.s3 nano-s3.gguf
bake.reference reference.wav
bake.gpu 0
bake.cond-seconds 15
bake.normalize-lufs -27
bake.trim-db 20
bake.voice-seconds 30
bake.prompt-seconds 10
bake.mel-seconds 10
bake.voice-rate 16000
bake.mel-rate 24000
bake.mel-fft 1920
bake.mel-hop 480
bake.mel-power 1
bake.mel-floor 1e-5
bake.mel-centered off
